   129→function timeLabel(iso: string): string {
   130→  const d = new Date(iso)
   131→  const now = new Date()
   132→  const sameDay = d.toDateString() === now.toDateString()
   133→  if (sameDay) {
   134→    return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
   135→  }
   136→  const yesterday = new Date(now)
   137→  yesterday.setDate(now.getDate() - 1)
   138→  if (d.toDateString() === yesterday.toDateString()) return 'Yesterday'
   139→  if (now.getTime() - d.getTime() < 6 * 24 * 3600 * 1000) {
   140→    return d.toLocaleDateString([], { weekday: 'short' })
   141→  }
   142→  return d.toLocaleDateString([], { day: 'numeric', month: 'short' })
   143→}

   159→function previewOf(c: Conversation, meId: string): string {
   160→  if (!c.lastMessage) return c.type === 'group' ? 'Group created' : 'Say hello'
   161→  const prefix = c.lastMessage.senderId === meId ? 'You: ' : c.type === 'group' ? `${c.lastMessage.senderName.split(' ')[0]}: ` : ''
   162→  const body = c.lastMessage.body.replace(/\s+/g, ' ')

   166→/** Escape a string for literal use inside a RegExp. */
   167→function escapeRe(s: string): string {
   168→  return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')

   171→/**
   172→ * Member-name mention matcher (word-bounded, case-insensitive, longest
   173→ * names first so "@Riya Sen" is consumed before a member named "Riya").
   174→ * Returns true if any matched token names the given user. Mirrors the
   175→ * server-side parseMentions rules in lib/chat.ts without importing it
   176→ * (lib/auth is server-only).
   177→ */
   178→function memberMentionTokens(body: string | undefined, members: ChatUser[]): string[] {
   179→  if (!body) return []
   180→  const names = members
   181→    .filter((m) => m.name && m.name.trim().length >= 2)
   182→    .map((m) => (m.name as string).trim())
   183→  if (!names.length) return []
   184→  const re = new RegExp(
   185→    `@(?:${[...names].sort((a, b) => b.length - a.length).map(escapeRe).join('|')})(?![a-zA-Z0-9_])`,
   186→    'gi'
   187→  )
   188→  const tokens: string[] = []
   189→  let mt: RegExpExecArray | null
   190→  while ((mt = re.exec(body)) !== null) tokens.push(mt[0].slice(1).toLowerCase())
   191→  return tokens
   192→}

   194→function mentionsMe(body: string | undefined, members: ChatUser[], myName: string): boolean {
   195→  const myLower = myName.trim().toLowerCase()
   196→  return Boolean(myLower) && memberMentionTokens(body, members).includes(myLower)
   197→}

   202→function isOnline(lastSeenAt?: string | null): boolean {
   203→  if (!lastSeenAt) return false
   204→  return Date.now() - new Date(lastSeenAt).getTime() < ONLINE_WINDOW_MS

   207→function lastSeenLabel(lastSeenAt?: string | null): string {
   208→  if (!lastSeenAt) return 'offline'
   209→  const delta = Date.now() - new Date(lastSeenAt).getTime()
   210→  if (delta < 60_000) return 'last seen just now'
   211→  if (delta < 3_600_000) return `last seen ${Math.floor(delta / 60_000)}m ago`
   212→  if (delta < 24 * 3_600_000) return `last seen ${Math.floor(delta / 3_600_000)}h ago`
   213→  if (delta < 48 * 3_600_000) return 'last seen yesterday'
   214→  return `last seen ${new Date(lastSeenAt).toLocaleDateString([], { day: 'numeric', month: 'short' })}`
   215→}

   217→/** window the body around the first case-insensitive match so the hit is visible */
   218→function snippetOf(body: string, q: string): string {
   219→  const clean = body.replace(/\s+/g, ' ')
   220→  const idx = clean.toLowerCase().indexOf(q.toLowerCase())
   221→  if (idx < 0) return clean.length > 120 ? clean.slice(0, 120) + '…' : clean
   222→  const start = Math.max(0, idx - 40)
   223→  const end = Math.min(clean.length, idx + q.length + 70)
   224→  return (start > 0 ? '…' : '') + clean.slice(start, end) + (end < clean.length ? '…' : '')
   225→}
