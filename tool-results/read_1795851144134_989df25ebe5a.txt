    61→   171→/**
    62→   172→ * Member-name mention matcher (word-bounded, case-insensitive, longest
    63→   173→ * names first so "@Riya Sen" is consumed before a member named "Riya").
    64→   174→ * Returns true if any matched token names the given user. Mirrors the
    65→   175→ * server-side parseMentions rules in lib/chat.ts without importing it
    66→   176→ * (lib/auth is server-only).
    67→   177→ */
    68→   178→function memberMentionTokens(body: string | undefined, members: ChatUser[]): string[] {
    69→   179→  if (!body) return []
    70→   180→  const names = members
    71→   181→    .filter((m) => m.name && m.name.trim().length >= 2)
    72→   182→    .map((m) => (m.name as string).trim())
    73→   183→  if (!names.length) return []
    74→   184→  const re = new RegExp(
    75→   185→    `@(?:${[...names].sort((a, b) => b.length - a.length).map(escapeRe).join('|')})(?![a-zA-Z0-9_])`,
    76→   186→    'gi'
    77→   187→  )
    78→   188→  const tokens: string[] = []
    79→   189→  let mt: RegExpExecArray | null
    80→   190→  while ((mt = re.exec(body)) !== null) tokens.push(mt[0].slice(1).toLowerCase())
    81→   191→  return tokens
    82→   192→}

   473→   583→          </span>
   474→   584→        )
   475→   585→        last = mt.index + token.length
   476→   586→      }
